# kia leaves out streams on your own network that come from a playlist from the internet
# (SECURITY.md says why); the word "lan" after the name lets a playlist have them.

# --- Radio ----------------------------------------------------------------------
# Internet radio stations from radio-browser.info, a free community directory. They
# appear under Radio in Kodi. Each line asks for the most popular working stations that match your choices:
#   country=US       two-letter country code
#   state=Texas      state or region (with country)
#   language=english
#   tag=jazz         one genre; tags=jazz,smooth for stations with all of them
#   station=KINK     stations whose name contains this
#   top=100          how many stations (default 100, most popular first)
#   min_bitrate=128  skip low-quality streams (kbps)
#   name=jazz        a name for the reports (otherwise made from the choices)
# radio-browser: country=US top=100                                 # 7,400 US stations to choose from
# radio-browser: country=US tag=jazz top=25
# radio-browser: country=US tag=news top=25
#
# Two stations that publish their own streams. SomaFM (somafm.com) is listener-supported
# and plays no ads; channels= picks some of its channels by the names in their addresses,
# such as somafm.com/groovesalad. KEXP is Seattle's listener-supported station, and its
# shows and DJs appear in the guide.
# somafm:                                                          # SomaFM, every channel     51
# somafm: channels=groovesalad,dronezone,secretagent name=somafm-chill
# kexp:                                                            # KEXP, with its show schedule
#
# More listener-supported stations, found by name in radio-browser (each one tested
# October 2026, all playing). top= is how many of the station's streams to take.
# radio-browser: station=paradise top=3 min_bitrate=128 name=radio-paradise  # Radio Paradise: Main, Rock, Mellow mixes
# radio-browser: station=wfmu top=3 name=wfmu                      # WFMU, New Jersey's freeform station
# radio-browser: station=kcrw top=2 name=kcrw                      # KCRW, Santa Monica: live and Eclectic 24
# radio-browser: station=kutx top=1 name=kutx                      # KUTX, Austin's music station
# radio-browser: station=nightride top=6 name=nightride            # Nightride FM's synthwave streams
# radio-browser: station=nts country=GB top=2 name=nts             # NTS Radio 1 and 2, London
# radio-browser: station=fip country=FR top=8 name=fip             # FIP and its themed streams, Paris
#
# Bigger genre lists, gathered from many station directories (github.com/junguler).
# Worldwide, names only (no logos). About half the stations worked in a test; kia keeps
# only those. The word "radio" at the end marks every entry as a radio station.
# https://raw.githubusercontent.com/junguler/m3u-radio-music-playlists/main/jazz.m3u   jazz   radio   # 403 stations, 210 working
# https://raw.githubusercontent.com/junguler/m3u-radio-music-playlists/main/80s.m3u    80s    radio   # 456 stations, 236 working
# Other genres: replace jazz.m3u with rock, classical, country, blues, 70s, 90s, ... (the
# list of files is at github.com/junguler/m3u-radio-music-playlists).
